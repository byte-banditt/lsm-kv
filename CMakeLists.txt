cmake_minimum_required(VERSION 3.16)
project(lsm_kv LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(lsm_kv src/codec.cpp src/wal.cpp src/skiplist.cpp src/sstable.cpp src/compactor.cpp src/db.cpp)
target_include_directories(lsm_kv PUBLIC include)
target_compile_options(lsm_kv PRIVATE -Wall -Wextra -Wpedantic)
add_executable(lsm_bench bench/bench.cpp)
target_link_libraries(lsm_bench PRIVATE lsm_kv)
include(CTest)
if(BUILD_TESTING)
  include(FetchContent)
  FetchContent_Declare(googletest URL https://github.com/google/googletest/archive/refs/tags/v1.15.2.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(googletest)
  add_executable(lsm_tests tests/tests.cpp)
  target_link_libraries(lsm_tests PRIVATE lsm_kv GTest::gtest_main)
  include(GoogleTest)
  gtest_discover_tests(lsm_tests)
endif()
